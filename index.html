// Microsserviço Distribuído com C# .NET 8, CQRS e Resiliência Polly
public sealed class ProcessFinancialRiskCommandHandler 
    : IRequestHandler<ProcessFinancialRiskCommand, Result<RiskAssessmentResponse>>
{
    private readonly IRiskEngineRepository _repository;
    private readonly IMessageBrokerPublisher _eventBus;
    private readonly IAsyncPolicy _resiliencePipeline;

    public async Task<Result<RiskAssessmentResponse>> Handle(
        ProcessFinancialRiskCommand cmd, CancellationToken ct)
    {
        return await _resiliencePipeline.ExecuteAsync(async () => 
        {
            // Executa validação de crédito e regras corporativas com concorrência otimista
            var score = await _repository.CalculateExposureAsync(cmd.AccountId, ct);
            await _eventBus.PublishAsync(new RiskEvaluatedEvent(cmd.AccountId, score), ct);
            return Result.Success(new RiskAssessmentResponse(score));
        });
    }
}
